# The exported Operator conformance suite.
#
# This is not a module and the autoloader never sees it: CPP_MODULE_ROOTS names
# modules/ only, and these translation units are compiled into a consuming
# repository's own executable rather than into a library here. The manifest
# exists so that scripts/check_modules.py has a node for the suite, because
# without one the suite was outside the module graph entirely -- first-party C++
# on a consumer's include path that could close a dependency cycle while the
# check still printed OK over a count that was never all the C++.
#
# There is no [module].type for the same reason. type names the kind of library
# cmake/build.cmake creates, and this file reaches no CMake at all:
# cmake/conformance-suite.cmake lists the suite's translation units by path and
# compiles them into the caller's executable. A type here would be a declaration
# nothing can honour -- which is what `type = sources` was, naming a kind the
# build had no branch for while the parser silently read it as `static`.
#
# The dependency list is what the suite's own headers and sources name, and it
# is public because every entry is named by include/conformance/, which a
# consumer compiles against. image and deployment are the two that are not
# reachable through operator: image is private to task and to vision, and
# operator must not depend on deployment -- an Operator that reached into a
# deployment for its own validators would be the deployment. So
# cmake/conformance-suite.cmake links both itself.

[module]
name = conformance
version = 0.1.0

[dependencies]
public = core deployment domain engine image operator script task trace
