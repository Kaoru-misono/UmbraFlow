# The Operator conformance suite, shipped as umbra-flow-conformance.
#
# This is not a module and the autoloader never sees it: CPP_MODULE_ROOTS names
# modules/ only, and conformance/CMakeLists.txt declares one executable rather
# than a library. The manifest exists so that scripts/check_modules.py has a
# node for the suite, because without one the suite was outside the module graph
# entirely -- first-party C++ that could close a dependency cycle while the
# check still printed OK over a count that was never all the C++.
#
# There is no [module].type for the same reason. type names the kind of library
# cmake/build.cmake creates, and this file reaches no CMake at all. A type here
# would be a declaration nothing can honour -- which is what `type = sources`
# was, naming a kind the build had no branch for while the parser silently read
# it as `static`.
#
# The dependency list is what the suite's own headers and sources name. It is
# public because include/conformance/ names every entry of it, and
# conformance/fixtures/ -- the last C++ project fixture, which tests/operator
# still includes -- compiles against the same headers. image and deployment are
# the two that are not reachable through operator: image is private to task and
# to vision, and operator must not depend on deployment -- an Operator that
# reached into a deployment for its own validators would be the deployment. So
# conformance/CMakeLists.txt links both itself.

[module]
name = conformance
version = 0.1.0

[dependencies]
public = core deployment domain engine image operator script task trace
