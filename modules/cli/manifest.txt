# The privileged annotation command surface: argument parsing, the queue/results
# protocol, and the process error/exit-code boundary. entry/ keeps only main.cpp,
# which is the executable's own translation unit.
#
# Privileged exploration is still implemented by task::ExplorationSession, so
# task and engine are public dependencies until the authoring host is split.

[module]
name = cli
type = static
version = 0.1.0

[build]
unity_build = false

# deployment carries the Operator, and this module is where the product binary
# reaches its registrar. It is private because open-project.hpp names no type
# from either: what a load produced is restated there as strings, so a caller
# reads the report without linking the authorities that answered for it.
#
# image and json are private for the same reason, and both arrived with the ocr
# verb. image decodes the PNG that verb reads and swizzles it into the channel
# order the recognition model was trained on; nothing but ocr-windows.cpp names
# image::RgbaImage, and ocr.hpp carries the decoded extent as two uint32 rather
# than the decoder's type. json is what formatImageText serializes through, and
# it appears in no header either: the verb returns std::string, because a caller
# of this module reads canonical bytes rather than a document model. json was
# already reachable through deployment, which is an accident of that edge and
# not a declaration this module can rely on.
#
# operator and trace arrived with the observe verb and are declared for that
# same reason. observe.cpp names OperatorCoordinator, because opening an
# already installed RuntimeArtifact is the only production mint of an
# activation authority, and it names TraceRecorder and FileTraceSink, because
# an EngineSession has nowhere else to record what it saw. Neither appears in
# observe.hpp: what the observation established is restated there as strings.
[dependencies]
public = engine task
private = deployment image json operator trace

# controller is public because targets.hpp and candidate-selection.hpp name its
# TargetCandidate, so a caller holding one has to see the type. user32 is
# private: only platform/ reaches the window API.
[dependencies.windows]
public = controller
private = user32

# The desktop half. Everything under platform/ is Windows API, and the three
# files beside it are the Windows implementations of headers whose declarations
# are platform-free.
[sources.windows]
include = candidate-selection.cpp candidate-selection.hpp explore-windows.cpp observe-windows.cpp ocr-windows.cpp platform/*.cpp platform/*.hpp targets-windows.cpp

# What answers the same declarations where there is no desktop to enumerate,
# and -- for ocr -- no inference runtime to load a model into.
[sources.other]
include = explore-unsupported.cpp observe-unsupported.cpp ocr-unsupported.cpp targets-unsupported.cpp
